-- Esquemas demo para explorar con DBA Manager.
-- Argumentos: &1 = password de los usuarios demo, &2 = nombre de la PDB
WHENEVER SQLERROR EXIT SQL.SQLCODE
SET ECHO OFF FEEDBACK OFF VERIFY OFF DEFINE ON

CREATE USER VENTAS   IDENTIFIED BY "&1" QUOTA UNLIMITED ON USERS;
CREATE USER RRHH     IDENTIFIED BY "&1" QUOTA UNLIMITED ON USERS;
CREATE USER BI_LECTOR IDENTIFIED BY "&1";
GRANT CREATE SESSION TO VENTAS, RRHH, BI_LECTOR;
-- Privilegios típicos de un esquema dueño (necesarios p. ej. para ser dueño de una vista materializada)
GRANT CREATE TABLE, CREATE VIEW, CREATE SEQUENCE, CREATE PROCEDURE, CREATE TRIGGER, CREATE MATERIALIZED VIEW TO VENTAS, RRHH;
GRANT CREATE SYNONYM TO BI_LECTOR;

-- ---------- VENTAS ----------
CREATE TABLE VENTAS.CLIENTES (
  ID_CLIENTE     NUMBER(10)     GENERATED BY DEFAULT AS IDENTITY,
  RFC            VARCHAR2(13)   NOT NULL,
  RAZON_SOCIAL   VARCHAR2(200)  NOT NULL,
  EMAIL          VARCHAR2(150),
  FECHA_ALTA     DATE           DEFAULT SYSDATE NOT NULL,
  ACTIVO         CHAR(1)        DEFAULT 'S' NOT NULL,
  CONSTRAINT PK_CLIENTES PRIMARY KEY (ID_CLIENTE),
  CONSTRAINT UK_CLIENTES_RFC UNIQUE (RFC),
  CONSTRAINT CK_CLIENTES_ACTIVO CHECK (ACTIVO IN ('S','N'))
);
COMMENT ON TABLE  VENTAS.CLIENTES IS 'Catálogo de clientes';
COMMENT ON COLUMN VENTAS.CLIENTES.RFC IS 'RFC del cliente';
COMMENT ON COLUMN VENTAS.CLIENTES.ACTIVO IS 'S = activo, N = inactivo';

CREATE TABLE VENTAS.PEDIDOS (
  ID_PEDIDO      NUMBER(12)     GENERATED BY DEFAULT AS IDENTITY,
  ID_CLIENTE     NUMBER(10)     NOT NULL,
  FECHA_PEDIDO   TIMESTAMP      DEFAULT SYSTIMESTAMP NOT NULL,
  ESTATUS        VARCHAR2(20)   DEFAULT 'NUEVO' NOT NULL,
  TOTAL          NUMBER(14,2)   DEFAULT 0 NOT NULL,
  CONSTRAINT PK_PEDIDOS PRIMARY KEY (ID_PEDIDO),
  CONSTRAINT FK_PEDIDOS_CLIENTE FOREIGN KEY (ID_CLIENTE) REFERENCES VENTAS.CLIENTES (ID_CLIENTE)
);
COMMENT ON TABLE VENTAS.PEDIDOS IS 'Encabezado de pedidos';
CREATE INDEX VENTAS.IX_PEDIDOS_CLIENTE ON VENTAS.PEDIDOS (ID_CLIENTE);
CREATE INDEX VENTAS.IX_PEDIDOS_FECHA ON VENTAS.PEDIDOS (FECHA_PEDIDO);

CREATE TABLE VENTAS.PEDIDO_DETALLE (
  ID_PEDIDO      NUMBER(12)     NOT NULL,
  RENGLON        NUMBER(5)      NOT NULL,
  SKU            VARCHAR2(40)   NOT NULL,
  CANTIDAD       NUMBER(10,3)   NOT NULL,
  PRECIO         NUMBER(14,2)   NOT NULL,
  CONSTRAINT PK_PEDIDO_DETALLE PRIMARY KEY (ID_PEDIDO, RENGLON),
  CONSTRAINT FK_DETALLE_PEDIDO FOREIGN KEY (ID_PEDIDO) REFERENCES VENTAS.PEDIDOS (ID_PEDIDO)
);

CREATE VIEW VENTAS.V_VENTAS_CLIENTE AS
  SELECT c.ID_CLIENTE, c.RAZON_SOCIAL, COUNT(p.ID_PEDIDO) PEDIDOS, SUM(p.TOTAL) TOTAL
    FROM VENTAS.CLIENTES c LEFT JOIN VENTAS.PEDIDOS p ON p.ID_CLIENTE = c.ID_CLIENTE
   GROUP BY c.ID_CLIENTE, c.RAZON_SOCIAL;

CREATE SEQUENCE VENTAS.SEQ_FOLIO START WITH 1000;

CREATE OR REPLACE PROCEDURE VENTAS.PRC_CERRAR_PEDIDO (p_id_pedido IN NUMBER) AS
BEGIN
  UPDATE VENTAS.PEDIDOS SET ESTATUS = 'CERRADO' WHERE ID_PEDIDO = p_id_pedido;
END;
/

-- ---------- RRHH ----------
CREATE TABLE RRHH.DEPARTAMENTOS (
  ID_DEPTO   NUMBER(6)     NOT NULL,
  NOMBRE     VARCHAR2(100) NOT NULL,
  CONSTRAINT PK_DEPARTAMENTOS PRIMARY KEY (ID_DEPTO)
);
CREATE TABLE RRHH.EMPLEADOS (
  ID_EMPLEADO  NUMBER(8)     NOT NULL,
  NOMBRE       VARCHAR2(120) NOT NULL,
  CURP         VARCHAR2(18),
  SALARIO      NUMBER(12,2),
  ID_DEPTO     NUMBER(6),
  CONSTRAINT PK_EMPLEADOS PRIMARY KEY (ID_EMPLEADO),
  CONSTRAINT FK_EMPLEADOS_DEPTO FOREIGN KEY (ID_DEPTO) REFERENCES RRHH.DEPARTAMENTOS (ID_DEPTO)
);
COMMENT ON TABLE RRHH.EMPLEADOS IS 'Plantilla de empleados (información sensible)';

-- ---------- BI_LECTOR: usuario lector con permisos y sinónimos ----------
GRANT SELECT ON VENTAS.CLIENTES TO BI_LECTOR;
GRANT SELECT ON VENTAS.PEDIDOS  TO BI_LECTOR;
GRANT SELECT ON VENTAS.V_VENTAS_CLIENTE TO BI_LECTOR;
CREATE SYNONYM BI_LECTOR.CLIENTES FOR VENTAS.CLIENTES;
CREATE SYNONYM BI_LECTOR.PEDIDOS  FOR VENTAS.PEDIDOS;

EXIT
